	// Funkce pro výpočet vzdálenosti mezi dvěma body pomocí haversinovy formule
	function calculateDistance2(lat1, lon1, lat2, lon2) {
		const R = 6371; // poloměr Země v km
		const dLat = deg2rad(lat2 - lat1);
		const dLon = deg2rad(lon2 - lon1);
		const a =
			Math.sin(dLat / 2) * Math.sin(dLat / 2) +
			Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) *
			Math.sin(dLon / 2) * Math.sin(dLon / 2);
		const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
		const d = R * c; // vzdálenost v km
		return d;
	}

	// Funkce pro převod stupňů na radiány
	function deg2rad(deg) {
		return deg * (Math.PI / 180);
	}
	
	function loadJson(lang) {
		if(jsonLoad == null){
			const cachedData = localStorage.getItem('cachedJSONData'+jsonLang);
            const lastFetchTime = localStorage.getItem('lastFetchTime'+jsonLang);
            const now = new Date().getTime();
            const oneDayInMilliseconds = 21600000;
			if (cachedData && lastFetchTime && (now - lastFetchTime) < oneDayInMilliseconds) {
				jsonData = JSON.parse(cachedData);
				//console.log(jsonData);
			} else {
				fetch(jsonFile)
					.then(response => response.json())
					.then(data => {
						// Vytvoření HTML obsahu z JSON dat
						jsonData = data;
						localStorage.setItem('cachedJSONData'+jsonLang, JSON.stringify(data));
                        localStorage.setItem('lastFetchTime'+jsonLang, now);
				})
				.catch(error => console.error('Error fetching JSON:', error));
			}
		}
	}

	async function displayClubs(geo = 1) {
		if (jsonLoad == null) {
			loadJson(jsonLang);
			jsonLoad = true;

						const currentLocation = await getCurrentLocation();
						const { latitude: currentLat, longitude: currentLong } = currentLocation.coords;

						const placesWithDistance = jsonData.map(place => {

							const distance = calculateDistance2(currentLat, currentLong, parseFloat(latitude), parseFloat(longitude));

						});
